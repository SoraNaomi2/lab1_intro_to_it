#include <iostream>
#include <cstdlib>
#include <ctime>
#include <algorithm>
using namespace std;
class numbers {
    int chis;
    int znam;

public:
    numbers(int x, int y) : chis(x), znam(y) {}
    numbers& operator+=(const numbers& other) {
        chis = chis * other.znam + other.chis * znam;
        znam = znam * other.znam;
        return *this;
    }

    void print() const {
        cout << chis << "/" << znam;
    }
};

int main() {
    srand(time(0));
    setlocale(LC_ALL, "ru");

    numbers a(1, 2);
    numbers b(1, 3); 
    numbers c(2, 5);  

    cout << " a += b:\n";
    a += b;
    cout << "a = "; a.print(); cout << endl;

    cout << "\na += c:\n";
    a += c;
    cout << "a = "; a.print(); cout << endl;

    cout << "\nb += c:\n";
    b += c;
    cout << "b = "; b.print(); cout << endl;

    return 0;
}
